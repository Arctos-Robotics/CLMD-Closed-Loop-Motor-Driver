cmake_minimum_required(VERSION 3.16)
project(clmd_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

function(clmd_case name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ../main stubs)
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -UNDEBUG)
        target_link_libraries(${name} PRIVATE m)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

clmd_case(core test_core.c ../main/core.c)
clmd_case(current test_current.c ../main/current.c)
clmd_case(joint test_joint.c ../main/core.c ../main/joint.c)
clmd_case(mt6701 test_mt6701.c ../main/mt6701_codec.c)
clmd_case(mt6701_i2c test_mt6701_i2c.c ../main/mt6701.c ../main/mt6701_codec.c)
